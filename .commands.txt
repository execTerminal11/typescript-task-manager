1) mkdir backend fontend
pnpm init -y


cd backend
2) Download base utiles
pnpm add -D typescript @types/node @/common

3) Generate the TypeScript configuration
pnpm exec tsc --init

4) update tsconfig.json
    "rootDir": "./src",
    "outDir": "./dist",
    "types": ["node"],


5) update package.json
  "scripts": {
    "test": "echo \"Error: no test specified\" && exit 1",
    "build": "tcs",
    "start": "node --import=tsx --watch dist/index.js"
  },



cd ../fontend/
6) fontend
pnpm create vite
pnpm add -D @types/react
pnpm add tailwindcss @tailwindcss/vite react


7) update tsconfig.json
{
  "compilerOptions": {
    "target": "es2023",
    "module": "esnext",
    "lib": ["ES2023", "DOM"],
    "types": ["vite/client"],
    "allowArbitraryExtensions": true,
    "skipLibCheck": true,
    "jsx": "react-jsx",
    "paths": {
      "@/*": ["./src/*"]
    },

    /* Bundler mode */
    "moduleResolution": "bundler",
    "allowImportingTsExtensions": true,
    "verbatimModuleSyntax": true,
    "moduleDetection": "force",
    "noEmit": true,

    /* Linting */
    "noUnusedLocals": true,
    "noUnusedParameters": true,
    "erasableSyntaxOnly": true,
    "noFallthroughCasesInSwitch": true
  },
  "include": ["src/**/*.ts", "src/**/*.d.ts", "src/**/*.tsx"]
}

8) update vite.config.ts
import { fileURLToPath, URL } from "node:url";
import react from "@vitejs/plugin-react";
import tailwindcss from "@tailwindcss/vite";
import { defineConfig } from "vite";

export default defineConfig({
  plugins: [react(), tailwindcss()],
  resolve: {
    alias: {
      "@": fileURLToPath(new URL("./src", import.meta.url))
    },
  },
});

9) update index.css
@import "tailwindcss";


cd ..
10) 
rm backend/pnpm-lock.yaml frontend/pnpm-lock.yaml
pnpm install
