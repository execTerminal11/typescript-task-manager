1) Download base utiles
pnpm add -D typescript @types/node @/common

2) Generate the TypeScript configuration
pnpm exec tsc --init

3) Configure ts configuration
    "rootDir": "./src",
    "outDir": "./dist",
    "types": ["node"],
enable in tsconfig.json

4) Run the project
pnpm exec tsc
node dist/index.js




cd fontend/
5) fontend
pnpm create vite
pnpm add -D @types/react
pnpm add tailwindcss @tailwindcss/vite react


6) update tsconfig.json
{
  "compilerOptions": {
    "target": "es2023",
    "module": "esnext",
    "lib": ["ES2023", "DOM"],
    "types": ["vite/client"],
    "allowArbitraryExtensions": true,
    "skipLibCheck": true,
    "jsx": "react-jsx",
    "paths": {
      "@/*": ["./src/*"]
    },

    /* Bundler mode */
    "moduleResolution": "bundler",
    "allowImportingTsExtensions": true,
    "verbatimModuleSyntax": true,
    "moduleDetection": "force",
    "noEmit": true,

    /* Linting */
    "noUnusedLocals": true,
    "noUnusedParameters": true,
    "erasableSyntaxOnly": true,
    "noFallthroughCasesInSwitch": true
  },
  "include": ["src/**/*.ts", "src/**/*.d.ts", "src/**/*.tsx"]
}

7) update vite.config.ts
import { fileURLToPath, URL } from "node:url";
import react from "@vitejs/plugin-react";
import tailwindcss from "@tailwindcss/vite";
import { defineConfig } from "vite";

export default defineConfig({
  plugins: [react(), tailwindcss()],
  resolve: {
    alias: {
      "@": fileURLToPath(new URL("./src", import.meta.url)),
      comp: fileURLToPath(new URL("./src/components", import.meta.url)),
      forth: fileURLToPath(
        new URL("./src/first/second/third/forth", import.meta.url),
      ),
    },
  },
});

8) update index.css
@import "tailwindcss";


cd project
pnpm init -y
